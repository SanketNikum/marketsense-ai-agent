# --- Stage 1: Data ingestion (no AI yet) ---
yfinance          # free EOD stock price data from Yahoo Finance
feedparser        # parses RSS news feeds (Moneycontrol, ET Markets)
python-dotenv     # loads secrets from .env into environment variables

# --- Stage 2: LLM classification ---
groq              # Groq's official client - fast, free-tier open model calls

# --- Stage 3: RAG (embeddings + vector storage) ---
sentence-transformers   # turns text into embeddings, runs locally, free
chromadb                # vector database - stores embeddings, does similarity search

# --- Stage 4: Agent orchestration ---
langgraph                # state machine framework for multi-step agent logic

# --- Stage 5: Tracing ---
langsmith                # tracing/observability dashboard for LangGraph runs
# Note: ragas was tried for automated eval scoring but conflicts with
# langgraph's dependency tree (needs pre-1.0 langchain). Skipped for now.

# --- Stage 6: API ---
fastapi                  # web framework - exposes the agent as an HTTP endpoint
uvicorn[standard]        # ASGI server that actually runs the FastAPI app
